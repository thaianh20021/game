	// "{15}" in a description is a per-level number: it is multiplied by the level shown
	public static string Desc(Boon b, int lvl) {
		var s = new System.Text.StringBuilder();
		string d = b.desc;
		for (int i = 0; i < d.Length; i++) {
			if (d[i] == '{') {
				int j = d.IndexOf('}', i);
				s.Append((int.Parse(d.Substring(i + 1, j - i - 1)) * lvl).ToString());
				i = j;
			} else s.Append(d[i]);
		}
		return s.ToString();
	}
